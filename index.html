    const scenesConfig = {};

    const btnListEl = document.getElementById('btn-list');

    const nodeKeys = Object.keys(campusNodes);


    function smoothMoveTo(arrow) {

      viewer.lookAt(arrow.pitch, arrow.yaw, 65, 350);

      setTimeout(() => {

        viewer.loadScene(arrow.target, arrow.targetPitch ?? 0, arrow.targetYaw ?? 0, 95);

      }, 320);

    }


    nodeKeys.forEach((key, idx) => {

      const node = campusNodes[key];

      scenesConfig[key] = {

        title: node.title,

        type: 'equirectangular',

        panorama: node.url,

        haov: 360,

        vaov: 105,

        hfov: 95,

        yaw: node.defaultYaw ?? 0,

        minPitch: -45,

        maxPitch: 45,

        hotSpots: node.arrows.map(arrow => ({

          pitch: arrow.pitch,

          yaw: arrow.yaw,

          type: 'scene',

          text: arrow.text,

          clickHandlerFunc: () => smoothMoveTo(arrow)

        }))

      };


      const btn = document.createElement('button');

      btn.className = 'map-btn' + (idx === 0 ? ' active' : '');

      btn.dataset.scene = key;

      btn.textContent = node.title;

      btn.onclick = () => viewer.loadScene(key, 0, node.defaultYaw ?? 0, 95);

      btnListEl.appendChild(btn);

    });


    const viewer = pannellum.viewer('panorama', {

      default: {

        firstScene: nodeKeys[0],

        sceneFadeDuration: 800,

        autoLoad: true

      },

      scenes: scenesConfig

    });


    viewer.on('scenechange', (sceneId) => {

      document.querySelectorAll('.map-btn').forEach(btn => {

        const isActive = btn.dataset.scene === sceneId;

        btn.classList.toggle('active', isActive);

        if (isActive) btn.scrollIntoView({ block: 'nearest', behavior: 'smooth' });

      });

    });


    const devHelper = document.getElementById('dev-helper');

    if (!DEV_MODE) {

      devHelper.style.display = 'none';

    } else {

      viewer.on('mousedown', (event) => {

        const coords = viewer.mouseEventToCoords(event);

        if (coords) {

          document.getElementById('click-angle').textContent =

            `yaw: ${Math.round(coords[1])}, pitch: ${Math.round(coords[0])}`;

        }

      });


      setInterval(() => {

        const curYaw = Math.round(viewer.getYaw());

        const curPitch = Math.round(viewer.getPitch());

      }, 200);

    }
